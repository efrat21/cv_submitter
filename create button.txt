Paste into PowerShell:

$WshShell = New-Object -ComObject WScript.Shell

# Find the actual Desktop folder
$Desktop = [Environment]::GetFolderPath('Desktop')
$ShortcutPath = Join-Path $Desktop 'Run Job Agent Daily.lnk'

# Create the shortcut. This runs the interactive workflow directly; the
# scheduled task remains reserved for unattended preparation and the digest.
$Shortcut = $WshShell.CreateShortcut($ShortcutPath)
$Shortcut.TargetPath = "$env:WINDIR\System32\cmd.exe"
$Shortcut.Arguments = '/c ""D:\cv_submitter\run_daily_review.bat""'
$Shortcut.WorkingDirectory = 'D:\cv_submitter'
$Shortcut.Description = 'Run the daily job preparation and review routine'
$Shortcut.Save()

Write-Host "Shortcut saved to: $ShortcutPath"
